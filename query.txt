"WITH input_macs AS (
    -- Reemplaza con tu lista de MACs
    SELECT unnest(ARRAY['54504C47B76EF250', '54504C47B76EAD60', '485754434EB78078', '54504C47B71CF9B0']::text[]) AS mac
),
mac_items AS (
    SELECT DISTINCT ON (m.mac)
        m.mac,
        i.hostid,
        replace(replace(i.name, 'ONU Serial ', ''), ' ', '') AS ont_id
    FROM input_macs m
    INNER JOIN history_text g ON g.value = m.mac
    INNER JOIN items i ON i.itemid = g.itemid
    INNER JOIN hosts h ON i.hostid = h.hostid
    WHERE i.key_ LIKE '%hwGponDeviceOntSn%'
      AND i.status = 0
      AND h.status = 0
      AND g.clock >= extract(epoch FROM now() - interval '30 days')::int
    ORDER BY m.mac, g.clock DESC
),
status_items AS (
    SELECT 
        m.mac,
        m.ont_id,
        i_stat.itemid AS status_itemid
    FROM mac_items m
    INNER JOIN items i_stat ON i_stat.hostid = m.hostid
    WHERE i_stat.key_ LIKE '%hwGponDeviceOntAlarmLOSi%'
      AND i_stat.status = 0
      AND replace(replace(i_stat.name, 'ONU LOSi ', ''), ' ', '') = m.ont_id
)
SELECT DISTINCT ON (s.status_itemid)
    s.mac,
    hs.value AS log,
    to_timestamp(hs.clock) AS time
FROM status_items s
INNER JOIN history_str hs ON hs.itemid = s.status_itemid
WHERE hs.clock >= extract(epoch FROM now() - interval '30 days')::int
ORDER BY s.status_itemid, hs.clock DESC;
"
